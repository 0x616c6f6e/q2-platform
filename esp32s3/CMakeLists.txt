cmake_minimum_required(VERSION 3.22)
set(Q2_KERNEL_DIR "" CACHE PATH "Independent Q2-Kernel directory")
if(NOT EXISTS "${Q2_KERNEL_DIR}/include/q2/kernel.h")
  message(FATAL_ERROR "Set Q2_KERNEL_DIR to a standalone Q2-Kernel checkout")
endif()
if(Q2_GENERATED_DIR)
  include("${Q2_GENERATED_DIR}/project.cmake")
endif()
option(Q2_ENABLE_CJK_FONT "Embed the Chinese font and enable Tiny TTF" ON)
set(IDF_TARGET esp32s3 CACHE STRING "SoC integration target")
if(NOT IDF_TARGET STREQUAL "esp32s3")
  message(FATAL_ERROR "This adapter is for ESP32-S3")
endif()
# Keep generated SDK state in the build tree; an old S31 sdkconfig may exist beside this port.
if(NOT SDKCONFIG)
  set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Generated board configuration")
endif()
set(PROJECT_VER "0.1.0-dev")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(q2_kernel_esp32s3)
