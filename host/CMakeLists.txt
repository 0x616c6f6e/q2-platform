cmake_minimum_required(VERSION 3.22)
project(Q2_Host LANGUAGES C)
set(Q2_KERNEL_DIR "" CACHE PATH "Independent Q2-Kernel source directory")
if(NOT EXISTS "${Q2_KERNEL_DIR}/include/q2/kernel.h")
  message(FATAL_ERROR "Pass -DQ2_KERNEL_DIR=/path/to/Q2-Kernel")
endif()
set(Q2_WITH_STORAGE ON CACHE BOOL "Storage")
set(Q2_BUILD_TESTS OFF CACHE BOOL "Integration tests")
include(../cmake/project-config.cmake)
add_subdirectory("${Q2_KERNEL_DIR}" kernel)
add_executable(q2_host main.c ../common/log_console.c)
target_include_directories(q2_host PRIVATE ../common)
target_link_libraries(q2_host PRIVATE q2_runtime q2_display q2_platform_config)

if(Q2_WITH_STORAGE)
  target_link_libraries(q2_host PRIVATE q2_storage)
endif()
if(Q2_WITH_SQLITE)
  target_link_libraries(q2_host PRIVATE q2_sqlite)
endif()
