cmake_minimum_required(VERSION 3.22)
project(Q2_X86 LANGUAGES C)
enable_testing()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The x86 GUI backend supports Windows or Linux (SDL2).")
endif()
set(Q2_KERNEL_DIR "" CACHE PATH "Independent Q2-Kernel source directory")
if(NOT EXISTS "${Q2_KERNEL_DIR}/include/q2/lvgl.h")
  message(FATAL_ERROR "Pass -DQ2_KERNEL_DIR=/path/to/Q2-Kernel with LVGL support")
endif()
set(Q2_WITH_LVGL ON CACHE BOOL "LVGL display")
set(Q2_WITH_STORAGE ON CACHE BOOL "Storage component")
set(Q2_BUILD_TESTS OFF CACHE BOOL "Kernel integration tests")
include(../cmake/project-config.cmake)
add_subdirectory("${Q2_KERNEL_DIR}" kernel)
if(WIN32)
  set(Q2_PC_SOURCES window_win32.c assets.c audio_win32.c network_win32.c)
  set(Q2_PC_LIBRARIES user32 gdi32 ws2_32 bcrypt winhttp winmm ole32)
  set(Q2_PC_STORAGE file_storage.c)
  set(Q2_PC_ASSET_TEST test_assets.c assets.c)
else()
  find_package(SDL2 2.0.18 REQUIRED CONFIG)
  find_package(CURL 7.68 REQUIRED)
  find_package(Threads REQUIRED)
  if(NOT Q2_GENERATED_DIR)
    # Standalone SDL2 builds disable the unimplemented audio backend.
    target_compile_definitions(q2_platform_config INTERFACE Q2_COMPONENT_AUDIO=0)
  endif()
  set(Q2_PC_SOURCES window_sdl2.c assets_posix.c network_curl.c)
  set(Q2_PC_LIBRARIES SDL2::SDL2 CURL::libcurl Threads::Threads)
  set(Q2_PC_STORAGE file_storage_posix.c)
  set(Q2_PC_ASSET_TEST test_assets_posix.c assets_posix.c)
endif()
add_executable(q2_x86 main.c pc_time.c device_data.c catalog.c web.c key_input.c
  ../common/log_console.c ${Q2_PC_SOURCES})
# Embed the development UI; the executable remains usable outside the source checkout.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/index.html")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/index.html" Q2_WEB_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," Q2_WEB_BYTES "${Q2_WEB_HEX}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/web_page.inc" "${Q2_WEB_BYTES}")
target_include_directories(q2_x86 PRIVATE ../common "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(q2_x86 PRIVATE c_std_11)
target_link_libraries(q2_x86 PRIVATE q2_runtime q2_lvgl q2_platform_config ${Q2_PC_LIBRARIES})
if(Q2_WITH_STORAGE)
  target_sources(q2_x86 PRIVATE ${Q2_PC_STORAGE})
  target_link_libraries(q2_x86 PRIVATE q2_storage)
endif()
if(Q2_WITH_SQLITE)
  target_link_libraries(q2_x86 PRIVATE q2_sqlite)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(q2_x86 PRIVATE -Wall -Wextra -Werror)
endif()

# Deterministic SoC gesture tests use a fake clock; no browser or wall-clock sleeps.
add_executable(q2_key_input_tests test_key_input.c key_input.c)
target_include_directories(q2_key_input_tests PRIVATE "${Q2_KERNEL_DIR}/include")
target_compile_features(q2_key_input_tests PRIVATE c_std_11)
# The algorithm suite uses a stable policy fixture, independent of user-edited product timings.
# Product header propagation is covered by root configuration and runtime integration checks.
target_include_directories(q2_key_input_tests PRIVATE ../common)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(q2_key_input_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME q2_key_input_tests COMMAND q2_key_input_tests)

# Exercise native path selection without requiring a window or image decoder.
add_executable(q2_assets_tests ${Q2_PC_ASSET_TEST})
target_include_directories(q2_assets_tests PRIVATE "${Q2_KERNEL_DIR}/include")
target_compile_features(q2_assets_tests PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(q2_assets_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME q2_assets_tests COMMAND q2_assets_tests)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(q2_sdl2_tests test_window_sdl2.c window_sdl2.c)
  target_include_directories(q2_sdl2_tests PRIVATE "${Q2_KERNEL_DIR}/include")
  target_compile_features(q2_sdl2_tests PRIVATE c_std_11)
  target_compile_options(q2_sdl2_tests PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(q2_sdl2_tests PRIVATE SDL2::SDL2)
  add_test(NAME q2_sdl2_tests COMMAND q2_sdl2_tests)
  set_tests_properties(q2_sdl2_tests PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND Q2_WITH_STORAGE)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME q2_posix_storage COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_storage_posix.py" --runner $<TARGET_FILE:q2_x86>)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_executable(q2_curl_tests test_network_curl.c network_curl.c pc_time.c)
  target_include_directories(q2_curl_tests PRIVATE "${Q2_KERNEL_DIR}/include")
  target_compile_features(q2_curl_tests PRIVATE c_std_11)
  target_compile_options(q2_curl_tests PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(q2_curl_tests PRIVATE CURL::libcurl Threads::Threads)
  add_test(NAME q2_curl_transport COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_network.py" --transport-runner $<TARGET_FILE:q2_curl_tests>)
  set_tests_properties(q2_curl_transport PROPERTIES TIMEOUT 30)
endif()
